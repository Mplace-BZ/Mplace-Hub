      // zostaje nadpisany przez autopilot i D[D.length-1] wskazuje na wrzesień zamiast październik.

      return [...rows].sort((a,b)=>{
        const ai=MONTHS.indexOf(a.m),bi=MONTHS.indexOf(b.m);

      });
